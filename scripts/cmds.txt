#!/bin/bash
# Enable persistence mode first
sudo nvidia-smi -pm 1

# Lock your desired graphics clock range (e.g., 208 to 2418 MHz)
sudo nvidia-smi --lock-gpu-clocks=208,2418

sudo nvidia-smi --lock-gpu-clocks=tdp,tdp

sudo nvidia-smi --lock-gpu-clocks=208,2150

# Follow these steps to create an automated, persistent boot service.

# Make the script executable by the system:
#    sudo chmod +x /usr/local/bin/gpu-persistent-settings.sh (this script)

# Create the Systemd Startup Service
# Now, configure Ubuntu/Linux to trigger this script automatically as soon as the NVIDIA drivers load during boot up.

#    1. Open a new systemd service file:
#    sudo nano /etc/systemd/system/nvidia-prop.service

#    2. Paste the following block into the file:

#    [Unit]
#    Description=Apply Custom NVIDIA GPU Power and Clock Limits
#    After=nvidia-persistenced.service
#    Wants=nvidia-persistenced.service

#    [Service]
#    Type=oneshot
#    RemainAfterExit=yes
#    ExecStart=/usr/local/bin/gpu-persistent-settings.sh

#    [Install]
#    WantedBy=multi-user.target

#    3. Save and close the file.

# Enable and Test the Service
# Tell the system to register and activate your new background service.

#    1. Reload the system configuration manager:
#    sudo systemctl daemon-reload

#    2. Enable the service so it automatically executes on all future reboots:
#    sudo systemctl enable nvidia-prop.service

#    3. Test-run the service immediately without restarting to ensure it works cleanly:
#    sudo systemctl start nvidia-prop.service

# To confirm it succeeded, verify the current service status by running `sudo systemctl status nvidia-prop.service`.
